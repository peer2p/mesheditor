<!-- Include Web Component script -->
<script src="mesheditor-element.bundle.js"></script>

<!-- Embed anywhere in your layout -->
<mesh-editor
  id="myEditor"
  src="/assets/models/character.glb"
  layout="stackpanel"
  grid="true"
  save-button-label="Save to Cloud"
  suppress-download="true"
></mesh-editor>
// Custom Plugin definition
const myPlugin = {
  id: 'my-studio-plugin',
  name: 'Studio Asset Sync',
  initialize(ctx) {
    // 1. Add top toolbar button
    ctx.registerToolbarButton({
      id: 'btnStudioSync',
      label: 'Sync',
      icon: 'upload-cloud',
      onClick: async () => {
        const glb = await ctx.exportGLB();
        await uploadToBackend(glb);
        ctx.showToast('Asset synchronized!', 'success');
      }
    });

    // 2. Add File / Tools dropdown item
    ctx.registerFileMenuItem({
      id: 'menuStudioDeploy',
      label: 'Deploy to Unreal Engine...',
      onClick: () => deployAsset()
    });
  }
};

// Register on editor ready
editor.addEventListener('editor-ready', async () => {
  await editor.usePlugin(myPlugin);
});
const editor = document.getElementById('myEditor');

// Intercept save button
editor.addEventListener('model-saved', (e) => {
  const { blob, filename } = e.detail;
  console.log(`Saved ${filename}: ${blob.size} bytes`);
});

// React to 3D selection changes
editor.addEventListener('selection-changed', (e) => {
  const { selectedMesh } = e.detail;
  updateInspector(selectedMesh?.name);
});

// React to scene modifications
editor.addEventListener('scene-modified', (e) => {
  const { meshCount, polyCount } = e.detail;
  console.log(`Scene: ${meshCount} meshes, ${polyCount} tris`);
});
